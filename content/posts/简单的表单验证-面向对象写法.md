## 面向对象的demo

最近一直在看关于面向对象部分的内容，也一直在思考，为什么程序的设计最终都倾向于选择这种变成思想。在逛遍了各种论坛，刷了各种知乎的答案。我觉得，可以归纳为几点。
1. 可复用
1. 低耦合，高内聚
1. 灵活的可扩展性

我更倾向于的理解是，通过这种设计，可以大大减少代码量，提高效率。

不废话了，看例子。有两个例子，都是通过亲手实践过的。为了更深层次地去理解这种写法，也是在边写的过程中，不断优化代码，不断思考。这两个例子，我用不同的方式写了三遍，也算是有了一些认识。

> 面向'类'的写法

```js
$(function($) {
    // 创建父类构造函数
    function Controler() {
        this.errors = [];
    }
    // 父类原型上添加显示信息方法
    Controler.prototype.showDialog = function(title, msg) {
        console.log(title, msg);
    }
    // 父类原型上添加success方法
    Controler.prototype.success = function(suc) {
        this.showDialog('Success', suc)
    }
    // 父类上添加failure方法
    Controler.prototype.failure = function(error) {
        this.errors.push(error)
        this.showDialog('Error', error)
    }
    // 创建LoginControler子类，处理登录时候的逻辑，获取用户名与密码，然后验证
    function LoginControler() {
        Controler.call(this)
    }
    // 创建关联，实现继承
    LoginControler.prototype = Object.create(Controler.prototype);
    // 获得用户名
    LoginControler.prototype.getUser = function() {
        return $('#user').val();
    }
    // 获取用户密码
    LoginControler.prototype.getPW = function() {
        return $('#password').val();
    }
    // 重写failure方法，当然不重写也是可以的，可以从原型中继承父级的方法
    LoginControler.prototype.failure = function(error) {
        Controler.prototype.failure.call(this, '登录失败：' + error);
    }
    // 前端拦截验证
    LoginControler.prototype.validateEntry = function(user, pw) {
        user = user || this.getUser();
        pw = pw || this.getPW();
        if (!(pw && user)) {
            return this.failure('用户名与密码不能为空')
        } else if (pw.length < 6) {
            return this.failure('密码长度不能小于6位')
        }
        return true;
    }
    // 创建AuthControler子类
    function AuthControler(login) {
        Controler.call(this);
        // 这个login实际上是LoginControler的一个实例
        this.login = login
    }
    // 实现继承
    AuthControler.prototype = Object.create(Controler.prototype);
    // 创建AJAX连接
    AuthControler.prototype.server = function(url, data) {
        return $.ajax({
            url: url,
            data: data,
            type: 'POST'
        })
    }
    // 重写success方法，同理，不重写当然也是可以的，这样是为了展示出可以灵活的自定义
    AuthControler.prototype.success = function() {
        Controler.prototype.success.call(this, '登录成功');
    }
    // 重写failuer方法
    AuthControler.prototype.failure = function() {
        Controler.prototype.failure.call(this, '登录失败');
    }
    // checkAuth方法
    AuthControler.prototype.checkAuth = function() {
        var user = this.login.getUser(),
            pw = this.login.getPW();
        if (this.login.validateEntry(user,pw)) {
            this.server('/checkAuth', {
                    user: user,
                    password: pw
                })
                .then(this.success.bind(this))
                .catch(this.failure.bind(this))
        }
    }
    $('#submit').click(function() {
        var auth = new AuthControler(new LoginControler());
        auth.checkAuth();
    })
})
```
针对上述代码，说一下构思。

父类的种种设计，都是为了子类。就如同，父母会努力赚钱，就为了子女上学，买房子等等。父类的各种属性和方法也是基于此。

所以我们创建了
* `showDialog`
* `success`
* `failure`

再说子类，子类如同子女，子女可以完全继承父母的一些特征，但是子女也是一个独立的个体，可以有自己的特征，然后也可以基于父母的特性形成自己的特征。同理，子类可以完全继承父类的方法，然后也可以改写（屏蔽）。

> 对象委托的方式（我称其为真正`面向对象`,因为就是利用对象，没有构造函数）

```js
$(function($) {
    var LoginControler = {
        // 错误列表
        errors: [],
        // 获取用户名
        getUser: function() {
            return $('#user').val();
        },
        // 获取密码
        getPassword: function() {
            return $('#password').val();
        },
        // 信息处理
        showDialog: function(title, msg) {
            console.log(title, msg)
        },
        // 错误处理
        failure: function(error) {
            this.errors.push(error);
            // 借用信息处理
            this.showDialog('Error', error);
        },
        // 前端表单验证
        validate: function(user, pw) {
            user = user || this.getUser();
            pw = pw || this.getPassword();
            if (!(user && pw)) {
                return this.failure('用户名或者密码不能为空')
            } else if (pw.length < 6) {
                return this.failure('密码至少6位')
            }
            return true;
        }
    }
    var AuthControler = Object.create(LoginControler);
    // 避免公用一个错误列表，所以新建一个
    AuthControler.errors = [];
    // 添加AJAX
    AuthControler.server = function(url, data) {
        return $.ajax({
            type: 'POST',
            url: url,
            data: data
        })
    }
    // success处理
    AuthControler.success = function() {
        // LoginControler.showDialog.call(this, '成功:', '验证通过');
        this.showDialog('成功', '验证通过')
    }
    // 错误处理
    AuthControler.rejecte = function(err) {
        // LoginControler.failure.call(this, '失败:' + err);
        this.failure('失败:' + err)
    }
    AuthControler.checkAuth = function() {
        var user = this.getUser();
        var pw = this.getPassword();
        if (this.validate(user, pw)) {
            this.server('/checkAuth', {
                    user: user,
                    password: pw
                })
                .then(this.success.bind(this))
                .catch(this.rejecte.bind(this, '验证未通过'))
        }
    }
    $('#submit').click(function() {
        var auth = Object.create(AuthControler);
        auth.checkAuth()
    })
})
```
    使用对象委托的形式
    使用对象委托形式，整个程序在设计上显得更为简洁，不拖泥带水
    也不需要一个父类，AuthControler可以直接继承LoginControler一些方法

这就是两个例子的写法。
