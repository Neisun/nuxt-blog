前一段时间使用Node.js中的express完成了文件上传工作。那是直接使用表单提交的原始方式，也就是说并没有利用ajax原理。这种方式缺点就是会使得表单提交之后页面有一个跳转，给用户的体验比较差劲。所以这一次的文件上传，我想用ajax来对表单进行提交，并且添加上进度条和上传文件的相关信息。
说说我们的思路：
首先，写好页面的布局。
其次，定义好事件，一个事件是input的onchange事件，用来获取上传文件的相关信息，然后展示出来做一个提示。
然后，另一个事件就是定义一个提交事件，即发送ajax请求
最后，需要开启一个服务器，不然无法上传。这里使用node中的express框架开启服务。

    // 如果是文件上传，那么form表单的enctype必须是multipart/form-data形式

**js部分**

js部分严格意义上分为两部分，其中一部分是node.js那部分，用来开启服务器的。另一部分是处理表单提交部分事件的js。

先看看处理表单那部分的js，直接用原生js写的，略显繁琐；
```javascript
var oFile = document.getElementById('file');
    var oFileName = document.getElementsByClassName('fileName')[0];
    var oFileType = document.getElementsByClassName('fileType')[0];
    var oFileSize = document.getElementsByClassName('fileSize')[0];
    var oSubmit = document.getElementById('submit');
    var oProgress = document.getElementById('progress');
    oFile.onchange = function() {
        var file = this.files[0];
        if (file) {
            // 做一个单位的转换
            var fileSize = 0;
            if (file.size > 1024 * 1024) {
                fileSize = (Math.round(file.size * 100 / (1024 * 1024)) / 100).toString() + "MB";
            } else {
                fileSize = (Math.round(file.size * 100 / 1024) / 100).toString() + "KB";
            }
            oFileName.innerHTML = '文件名：' + file.name;
            oFileType.innerHTML = '类  型：' + file.type;
            oFileSize.innerHTML = '大  小：' + fileSize;
        }
    }
    oSubmit.onclick = function () {
        // 创建formData用来储存表单的数据
        var formData = new FormData(document.getElementById('uploadForm'));
        console.log(formData);
        // 实例化ajax
        // 这是一个最简单的方式，当然压根就没有做好兼容性；
        // 推荐使用jq中的ajax，封装的比较到位
        var xhr = new XMLHttpRequest();
        xhr.onload = function () {
            console.log('load')
        }
        // 文件上传progress事件
        xhr.upload.onprogress = function (e) {
            console.log(e);
            if (e.lengthComputable) {
                oProgress.value = e.loaded;
                oProgress.max = e.total;
            }else {
                oProgress.innerHTML = '无法计算';
            }
            console.log('progress')
        }
        xhr.onerror = function () {
            console.log('error')
        }
        xhr.onabort = function () {
            console.log('abort')
        }
        xhr.open('post','/upload',true);
        xhr.send(formData);
    }
```

那么另外一部分js文件就是利用express框架开启本地服务器，并且设置好post请求。

```javascript
/*
 * @Author: hunaisong 
 * @Date: 2017-10-10 16:19:33 
 * @Last Modified by: hunaisong

// 需要的依赖
var express = require('express'),
    app = express(),
    fs = require('fs'),
    path = require('path'),
    // body-parser - node.js 中间件，用于处理 JSON, Raw, Text 和 URL 编码的数据。
    bodyParser = require('body-parser'),
    // multer - node.js 中间件，用于处理 enctype="multipart/form-data"（设置表单的MIME编码）的表单数据。
    multer = require('multer');
// 使用中间件 dest应该是存在内存中的位置
app.use(multer({ dest: '/tem/' }).array('file'));
// 使用静态文件，如果需要加载jquery
app.use(express.static('public'))
app.get('/uploadFile.html', (req, res) => {
    res.sendfile(path.join(__dirname, 'uploadFile.html'))
})
app.post('/upload', (req, res) => {
    console.log(req.files[0]);
    // 设置一个assets文件夹存放上传的内容
    var dest = __dirname + '/assets/' + req.files[0].originalname;
    // 先读取上传的文件
    fs.readFile(req.files[0].path, (err, data) => {
        if (err) {
            console.log(err)
        } else {
            // 将读取的文件写入assets文件夹
            fs.writeFile(dest, data, (err) => {
                if (err) {
                    console.log(err)
                } else {
                    response = {
                        message: 'File upload successfully!',
                        filename: req.files[0].originalname
                    }
                }
                console.log(response);
                res.writeHead(200, { 'Content-Type': 'charset=utf8' });
                res.end(JSON.stringify(response))
            })
        }
    })
})
var server = app.listen(8888, () => {

})
```

总结一下，通过这个demo，我们能从中得到什么知识点：
1.文件上传，form的enctype是multipart/form-data
2.获取上传文件的信息，在`<input type="file">`上绑定的onchange事件上来触发
3.进度事件，xhr的事件，具体事件需要参考那篇博客，说的很详细。咱们这里因为是上传，所以是xhr.upload.onprogress事件
4.开启你的本地服务器
